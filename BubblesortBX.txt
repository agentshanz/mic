.MODEL SMALL
.STACK 100H

.DATA
ARRAY DW 0012H,0005H,0018H,0003H,000AH
COUNT EQU 5

.CODE
MAIN PROC

    MOV AX,@DATA
    MOV DS,AX

    MOV BP,COUNT-1

OUTER:
    LEA BX,ARRAY
    MOV CX,BP

INNER:
    MOV AX,[BX]
    CMP AX,[BX+2]
    JBE SKIP

    MOV DX,[BX+2]
    MOV [BX+2],AX
    MOV [BX],DX

SKIP:
    ADD BX,2
    LOOP INNER

    DEC BP
    JNZ OUTER

MAIN ENDP
END MAIN
