.MODEL SMALL
.STACK 100H
.DATA
ARRAY DW 0012H,0005H,0018H,0003H,000AH
COUNT EQU 5

.CODE
MAIN PROC

    MOV AX,@DATA
    MOV DS,AX

    MOV BP,COUNT-1

OUTER:
    LEA SI,ARRAY
    MOV CX,BP

INNER:
    MOV AX,[SI]
    CMP AX,[SI+2]
    JBE SKIP

    MOV DX,[SI+2]
    MOV [SI+2],AX
    MOV [SI],DX

SKIP:
    ADD SI,2
    LOOP INNER

    DEC BP
    JNZ OUTER

MAIN ENDP
END MAIN
