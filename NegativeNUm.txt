.model small
.stack 100h

.data
NUMBERS DB 25,-10,15,-5,-20,30,-8,12
N EQU 8
COUNT DB 0

MSG DB 13,10,'Number of negative numbers = $'

.code
main PROC

    mov ax,@data
    mov ds,ax

    lea si,NUMBERS
    mov cx,N
    mov bl,00H

CHECK:
    mov al,[si]

    cmp al,00H
    jge NEXT

    inc bl

NEXT:
    inc si
    loop CHECK

    mov COUNT,bl

    ; Display message
    mov dx,offset MSG
    mov ah,09H
    int 21H

    ; Display count
    mov dl,COUNT
    add dl,'0'
    mov ah,02H
    int 21H
main ENDP
end main
