org 100h

#start=Traffic_Lights.exe#

; Initial all-red condition
mov ax, all_red
out 4, ax

; Point to first traffic-light pattern
mov si, offset start

next:
    ; Get traffic-light pattern
    mov ax, [si]

    ; Send pattern to Traffic Lights
    out 4, ax

    ; Approximately 8 seconds delay
    mov cx, 70h
    mov dx, 0000h
    mov ah, 86h
    int 15h

    ; Move to next pattern
    add si, 2

    ; Check whether all patterns are completed
    cmp si, s_end
    jb next

    ; Start again from first pattern
    mov si, offset start
    jmp next


; FEDC_BA98_7654_3210

start dw 0000_0010_0100_1001b

s1    dw 0000_0010_0100_1010b

s2    dw 0000_0010_0100_1100b

s3    dw 0000_0010_0101_0001b

s4    dw 0000_0010_0110_0001b

s5    dw 0000_0010_1000_1001b

s6    dw 0000_0011_0000_1001b

s7    dw 0000_0100_0100_1001b

s8    dw 0000_1000_0100_1001b

s9    dw 0000_0011_0000_1100b

s11   dw 0000_0110_1001_1010b

s12   dw 0000_1000_0110_0001b

s_end = $

all_red equ 0000_0010_0100_1001b
